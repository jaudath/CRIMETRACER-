from datetime import datetime, timedelta
from jose import jwt, JWTError
from passlib.context import CryptContext
from .config import settings

pwd_context = CryptContext(schemes=["bcrypt"], deprecated="auto")
ALGORITHM = "HS256"

def verify_password(plain_password, hashed_password):
    return pwd_context.verify(plain_password, hashed_password)

def get_password_hash(password):
    return pwd_context.hash(password)

def create_access_token(data: dict, expires_delta: timedelta = None):
    to_encode = data.copy()
    expire = datetime.utcnow() + (expires_delta or timedelta(minutes=settings.ACCESS_TOKEN_EXPIRE_MINUTES))
    to_encode.update({"exp": expire})
    encoded_jwt = jwt.encode(to_encode, settings.SECRET_KEY, algorithm=ALGORITHM)
    return encoded_jwt

def decode_token(token: str):
    try:
        payload = jwt.decode(token, settings.SECRET_KEY, algorithms=[ALGORITHM])
        return payload
    except JWTError:
        return None

def generate_precautions(crime_type: str, complaint_text: str, locale: str="en"):
    prompt = f"""You are a Chennai Police safety assistant.
A citizen reported: {crime_type}
Complaint details: {complaint_text}
Provide 3-4 short actionable precaution steps the victim can take now to stay safe and preserve evidence.
Return a JSON array of strings only.
"""
    if not GEMINI_API_KEY:
        # simple fallback
        return {"precautions": ["Do not return to the scene alone.", "Preserve evidence (photos/screenshots).", "Share location with trusted contact."], "raw": "fallback"}
    # Real API call (replace with actual spec for Gemini)
    url = "https://api.gemini.example/v1/generate"
    headers = {"Authorization": f"Bearer {GEMINI_API_KEY}", "Content-Type": "application/json"}
    resp = requests.post(url, json={"prompt": prompt, "max_tokens": 256, "temperature": 0.2}, headers=headers, timeout=10)
    if resp.status_code != 200:
        return {"precautions": [], "raw": resp.text}
    text = resp.json().get("text") or resp.text
    # naive extraction
    items = [s.strip() for s in text.split("\n") if s.strip()]
    return {"precautions": items[:4], "raw": text}

import joblib
import xgboost as xgb
import pandas as pd
import numpy as np
from scipy.sparse import hstack, csr_matrix
from pathlib import Path

# Base paths
BASE_DIR = Path(__file__).resolve().parent.parent
MODEL_DIR = BASE_DIR / "models"

MODEL_PATH = MODEL_DIR / "xgb_classifier.model"
PREPROCESSORS_PATH = MODEL_DIR / "preprocessors_cls.joblib"

# Severity mappings
sev_map = {"low": 0, "medium": 1, "high": 2}
sev_map_rev = {0: "Low", 1: "Medium", 2: "High"}
officer_map = {0: 1, 1: 5, 2: 10}

# Rule-based crime to severity mapping (dataset consistency)
crime_to_severity = {
    "theft": "low", "robbery": "medium", "burglary": "medium",
    "assault": "medium", "dispute": "medium", "fraud": "medium",
    "murder": "high", "homicide": "high", "kidnapping": "high", "rape": "high",
}

# Load model + preprocessors once at startup
model = xgb.XGBClassifier()
model.load_model(str(MODEL_PATH))

preprocessors = joblib.load(PREPROCESSORS_PATH)
tfidf = preprocessors["tfidf"]
ohe = preprocessors["ohe"]
scaler = preprocessors["scaler"]
loc_counts_map = preprocessors["loc_counts"]

def predict_new_complaint(complaint_text, crime_type, location, timestamp, severity=None):
    """Predict officers required given a new complaint."""

    # --- Normalize inputs ---
    complaint_text = "" if complaint_text is None else str(complaint_text).lower().strip()
    crime_type = "Unknown" if not crime_type else str(crime_type).title().strip()
    location = "unknown" if not location else str(location).lower().strip()

    complaint_len = len(complaint_text)
    ts_parsed = pd.to_datetime(timestamp, errors="coerce")
    report_hour = ts_parsed.hour if pd.notna(ts_parsed) else -1

    # --- Decide severity ---
    # Priority: provided severity > dataset crime_type mapping > default "medium"
    sev_final = None
    if severity:
        sev_final = severity.lower()
    else:
        sev_final = crime_to_severity.get(crime_type.lower(), "medium")

    severity_num = sev_map.get(sev_final, 1)
    loc_freq = loc_counts_map.get(location, 1)

    # --- Feature construction ---
    X_text_new = tfidf.transform([complaint_text])
    X_crime_new = ohe.transform([[crime_type]])
    X_num_new = np.array([[complaint_len, report_hour, severity_num, loc_freq]])
    X_num_new_scaled = scaler.transform(X_num_new)

    X_new = hstack([X_text_new, X_crime_new, csr_matrix(X_num_new_scaled)], format="csr")

    # --- Model prediction ---
    pred_class = int(model.predict(X_new)[0])
    pred_category = sev_map_rev[pred_class]

    # --- Ensure logical consistency ---
    # If model predicts "Medium" but rules say "High" (e.g., Murder), override
    if sev_map[sev_final] > pred_class:
        pred_class = sev_map[sev_final]
        pred_category = sev_map_rev[pred_class]

    officers_required = officer_map[pred_class]

    return {
        "officers_required": officers_required,
        "requirement_category": pred_category,
        "predicted_class": pred_class,
        "rule_based_severity": sev_final.title()
    }

#!/usr/bin/env python3
"""
Optimized training script for CrimeTracer officer requirement model.

- Loads victim_complaints.csv and case_tracking.csv from given dataset folder
- Optionally uses Chennai_Crime_Data_Fixed.csv (if it exists) to correct severity labels
- Removes label leakage (DO NOT include severity_num as a feature)
- Balances classes in-memory (upsampling) before training
- Trains XGBoost (fallback to DummyClassifier if xgboost missing)
- Saves:
    ./models/xgb_classifier.model      (xgboost native save if available)
    ./models/xgb_classifier.pkl        (joblib dump of model object)
    ./models/preprocessors_cls.joblib  (tfidf, ohe, scaler, loc_counts)
"""

MODEL_PATH_XGB = OUT_DIR / "xgb_classifier.model"
MODEL_PATH_PKL = OUT_DIR / "xgb_classifier.pkl"
PREPROCESSORS_PATH = OUT_DIR / "preprocessors_cls.joblib"

SEED = 42
TEST_SIZE = 0.2
TFIDF_MAX_FEATURES = 3000  # increase vocab a bit

# label maps
crime_to_severity_default = {
    "theft": "low", "robbery": "medium", "burglary": "medium",
    "assault": "medium", "dispute": "medium", "fraud": "medium",
    "murder": "high", "homicide": "high", "kidnapping": "high", "rape": "high",
}
sev_map = {"low": 0, "medium": 1, "high": 2}
sev_map_rev = {0: "Low", 1: "Medium", 2: "High"}
officer_map = {0: 1, 1: 5, 2: 10}

# logging
logging.basicConfig(level=logging.INFO, format="%(asctime)s | %(levelname)s | %(message)s")
log = logging.getLogger("train_xgb")

def safe_onehot_encoder():
    """Return OneHotEncoder compatible with various sklearn versions."""
    try:
        return OneHotEncoder(handle_unknown="ignore", sparse_output=True)
    except TypeError:
        return OneHotEncoder(handle_unknown="ignore", sparse=True)

def require_file(p: Path):
    if not p.exists():
        raise FileNotFoundError(f"Required file not found: {p}")

def load_csv(path: Path):
    require_file(path)
    df = pd.read_csv(path)
    # normalize columns lower & strip
    df.columns = [c.strip() for c in df.columns]
    return df

def main():
    try:
        log.info("Python %s | Platform: %s", sys.version.split()[0], platform.platform())
        log.info("Datasets dir: %s", DATASETS_DIR)

        # load victim complaints & case tracking
        require_file(VC_PATH)
        require_file(CASE_PATH)
        vc = pd.read_csv(VC_PATH)
        case = pd.read_csv(CASE_PATH)
        log.info("Loaded victim complaints: %d rows", len(vc))
        log.info("Loaded case tracking:       %d rows", len(case))

        # Normalize column names to lower for matching convenience
        vc.columns = [c.lower().strip() for c in vc.columns]
        case.columns = [c.lower().strip() for c in case.columns]

        # Optional: if Chennai dataset exists, load and use its severity as source of truth
        if CHENNAI_PATH.exists():
            ch = pd.read_csv(CHENNAI_PATH)
            ch.columns = [c.lower().strip() for c in ch.columns]
            log.info("Chennai reference dataset found (%s). Using it to overwrite severity where available.", CHENNAI_PATH.name)
            # we expect ch to have complaint_id and severity (per your earlier schema)
            if "complaint_id" in ch.columns and "severity" in ch.columns:
                # merge by complaint_id: victim complaints has complaint_id
                vc = vc.merge(ch[["complaint_id", "severity"]].rename(columns={"severity": "severity_chennai"}), on="complaint_id", how="left")
                # if ch provides severity, force it; else keep existing severity (or infer later)
                vc["severity"] = vc["severity_chennai"].fillna(vc.get("severity", ""))
                vc.drop(columns=["severity_chennai"], inplace=True)
                log.info("Merged severity from Chennai for %d rows.", vc["severity"].notnull().sum())
            else:
                log.warning("Chennai file present but missing expected columns ('complaint_id','severity'). Skipping Chennai merge.")
        else:
            log.info("Chennai reference dataset not found. Proceeding without external severity correction.")

        # Normalize fields we need and canonicalize text
        for col in ["complaint_text", "location", "crime_type", "severity"]:
            if col in vc.columns:
                vc[col] = vc[col].fillna("").astype(str).str.strip()
            else:
                vc[col] = ""

        # Lowercase location for counts; crime_type title-case for OHE consistency
        vc["complaint_text"] = vc["complaint_text"].str.lower()
        vc["location"] = vc["location"].str.lower()
        vc["crime_type"] = vc["crime_type"].str.title()
        vc["severity"] = vc["severity"].str.lower()

        # If severity blank, infer from mapping
        vc["severity"] = vc.apply(lambda r: crime_to_severity_default.get(r["crime_type"].lower(), r["severity"] or "medium"), axis=1)

        # Make sure final labels are in allowed set; default to medium if unexpected
        vc["severity"] = vc["severity"].map(lambda s: s if s in sev_map else "medium")

        # Remove label leakage: do not create/use severity_num as feature
        # Create numeric target y
        y = vc["severity"].map(sev_map).astype(int)

        # Basic EDA logs
        log.info("Final severity distribution (raw): %s", vc["severity"].value_counts().to_dict())

        # Feature engineering
        vc["complaint_len"] = vc["complaint_text"].str.len()
        # Parse timestamp hour if possible
        if "timestamp" in vc.columns:
            vc["timestamp_parsed"] = pd.to_datetime(vc["timestamp"], errors="coerce")
            vc["report_hour"] = vc["timestamp_parsed"].dt.hour.fillna(-1).astype(int)
        else:
            vc["report_hour"] = -1

        # Location frequency count (helps represent hotspot)
        loc_counts = vc["location"].value_counts().to_dict()
        vc["loc_freq"] = vc["location"].map(lambda l: loc_counts.get(l, 1)).fillna(1).astype(int)

        # Build numeric array WITHOUT severity_num
        numeric_cols = ["complaint_len", "report_hour", "loc_freq"]
        X_num = vc[numeric_cols].fillna(0).values

        # Text TF-IDF
        tfidf = TfidfVectorizer(max_features=TFIDF_MAX_FEATURES, stop_words="english")
        X_text = tfidf.fit_transform(vc["complaint_text"].astype(str))

        # OneHot encode crime_type
        ohe = safe_onehot_encoder()
        X_crime = ohe.fit_transform(vc[["crime_type"]])

        # Scaler for numeric
        scaler = StandardScaler(with_mean=False)
        X_num_scaled = scaler.fit_transform(X_num)

        # Combine sparse matrices
        X = hstack([X_text, X_crime, csr_matrix(X_num_scaled)], format="csr")
        log.info("Combined feature matrix shape: %s", X.shape)

        # Balance classes in-memory by upsampling minority classes (simple strategy)
        df_features = vc.copy()
        df_features["y"] = y
        class_counts = df_features["y"].value_counts()
        max_count = class_counts.max()
        balanced_parts = []
        for cls, cnt in class_counts.items():
            subset = df_features[df_features["y"] == cls]
            if cnt == 0:
                continue
            # repeat full blocks
            times = max_count // cnt
            rem = max_count % cnt
            if times > 0:
                balanced_parts.append(pd.concat([subset] * times, ignore_index=True))
            if rem > 0:
                balanced_parts.append(subset.sample(rem, replace=True, random_state=SEED))
        df_balanced = pd.concat(balanced_parts, ignore_index=True).sample(frac=1, random_state=SEED)
        log.info("Balanced dataset rows: %d (per-class max %d)", len(df_balanced), max_count)

        # Build balanced X and y consistent with feature pipeline
        # Note: rebuilding X for balanced df is simpler and safer
        # Recompute text/crime_type/numeric for df_balanced
        tfidf_bal = tfidf  # fitted already
        ohe_bal = ohe
        scaler_bal = scaler

        X_text_b = tfidf_bal.transform(df_balanced["complaint_text"].astype(str))
        X_crime_b = ohe_bal.transform(df_balanced[["crime_type"]])
        X_num_b = scaler_bal.transform(df_balanced[numeric_cols].fillna(0).values)
        X_bal = hstack([X_text_b, X_crime_b, csr_matrix(X_num_b)], format="csr")
        y_bal = df_balanced["y"].astype(int).values

        log.info("Train/test split: rows=%d", X_bal.shape[0])
        stratify_param = y_bal if len(np.unique(y_bal)) > 1 else None
        X_train, X_test, y_train, y_test = train_test_split(X_bal, y_bal, test_size=TEST_SIZE, random_state=SEED, stratify=stratify_param)

        log.info("Train shape: %s | Test shape: %s", X_train.shape, X_test.shape)

        # Train XGBoost if available
        model_obj = None
        used_xgb = False
        try:
            xgb = importlib.import_module("xgboost")
            log.info("Using XGBoost version %s", getattr(xgb, "__version__", "unknown"))
            clf = xgb.XGBClassifier(
                n_estimators=200,
                max_depth=6,
                learning_rate=0.08,
                subsample=0.8,
                colsample_bytree=0.8,
                random_state=SEED,
                tree_method="hist",
                objective="multi:softprob",  # allow probabilities
                num_class=len(np.unique(y_train)),
                use_label_encoder=False,
                eval_metric="mlogloss"
            )
            clf.fit(X_train, y_train, eval_set=[(X_test, y_test)], verbose=False)
            model_obj = clf
            used_xgb = True
        except Exception as e:
            log.warning("XGBoost training failed or not installed: %s. Falling back to DummyClassifier.", e)
            dummy = DummyClassifier(strategy="most_frequent")
            dummy.fit(X_train, y_train)
            model_obj = dummy

        # evaluate
        y_pred = model_obj.predict(X_test)
        log.info("Classification report:\n%s", classification_report(y_test, y_pred, target_names=["Low", "Medium", "High"], zero_division=0))
        log.info("Accuracy: %.4f", accuracy_score(y_test, y_pred))

        # Save preprocessors (tfidf, ohe, scaler, loc_counts)
        preprocessors = {
            "tfidf": tfidf,
            "ohe": ohe,
            "scaler": scaler,
            "loc_counts": loc_counts
        }
        joblib.dump(preprocessors, PREPROCESSORS_PATH)
        log.info("Saved preprocessors to %s", PREPROCESSORS_PATH)

        # Save model: both xgb native and joblib fallback
        if used_xgb and hasattr(model_obj, "save_model"):
            try:
                model_obj.save_model(str(MODEL_PATH_XGB))
                log.info("Saved XGBoost model to %s", MODEL_PATH_XGB)
            except Exception as e:
                log.warning("Could not save xgboost model with save_model(): %s", e)
        joblib.dump(model_obj, MODEL_PATH_PKL)
        log.info("Saved model (joblib) to %s", MODEL_PATH_PKL)

        # quick sanity predictions
        sample_n = min(3, X_test.shape[0])
        for i in range(sample_n):
            xi = X_test[i]
            pc = model_obj.predict(xi)
            log.info("Sample %d -> class %s -> officers %s", i, int(pc[0]), officer_map.get(int(pc[0]), 1))

        log.info("Training complete. Artifacts in: %s", OUT_DIR.resolve())

    except Exception as e:
        log.error("Training script failed: %s", e)
        traceback.print_exc()
        sys.exit(1)


if __name__ == "__main__":
    main()

src/contexts/ThemeContext.tsx
import React, { createContext, useContext, useEffect, useState } from 'react';

type Theme = 'light' | 'dark';

interface ThemeContextType {
  theme: Theme;
  toggleTheme: () => void;
}

const ThemeContext = createContext<ThemeContextType | undefined>(undefined);

export const useTheme = () => {
  const context = useContext(ThemeContext);
  if (!context) {
    throw new Error('useTheme must be used within a ThemeProvider');
  }
  return context;
};

interface ThemeProviderProps {
  children: React.ReactNode;
}

export const ThemeProvider: React.FC<ThemeProviderProps> = ({ children }) => {
  const [theme, setTheme] = useState<Theme>(() => {
    // Check localStorage first, then system preference
    const savedTheme = localStorage.getItem('crimetracer-theme') as Theme;
    if (savedTheme) {
      return savedTheme;
    }
    return window.matchMedia('(prefers-color-scheme: dark)').matches ? 'dark' : 'light';
  });

  useEffect(() => {
    const root = document.documentElement;
    root.classList.remove('light', 'dark');
    root.classList.add(theme);
    localStorage.setItem('crimetracer-theme', theme);
  }, [theme]);

  const toggleTheme = () => {
    setTheme(prev => prev === 'light' ? 'dark' : 'light');
  };

  return (
    <ThemeContext.Provider value={{ theme, toggleTheme }}>
      {children}
    </ThemeContext.Provider>
  );
};

import * as React from "react";

const MOBILE_BREAKPOINT = 768;

export function useIsMobile() {
  const [isMobile, setIsMobile] = React.useState<boolean | undefined>(undefined);

  React.useEffect(() => {
    const mql = window.matchMedia(`(max-width: ${MOBILE_BREAKPOINT - 1}px)`);
    const onChange = () => {
      setIsMobile(window.innerWidth < MOBILE_BREAKPOINT);
    };
    mql.addEventListener("change", onChange);
    setIsMobile(window.innerWidth < MOBILE_BREAKPOINT);
    return () => mql.removeEventListener("change", onChange);
  }, []);

  return !!isMobile;
}

src/hooks/use-toast.ts
import * as React from "react";

import type { ToastActionElement, ToastProps } from "@/components/ui/toast";

const TOAST_LIMIT = 1;
const TOAST_REMOVE_DELAY = 1000000;

type ToasterToast = ToastProps & {
  id: string;
  title?: React.ReactNode;
  description?: React.ReactNode;
  action?: ToastActionElement;
};

const actionTypes = {
  ADD_TOAST: "ADD_TOAST",
  UPDATE_TOAST: "UPDATE_TOAST",
  DISMISS_TOAST: "DISMISS_TOAST",
  REMOVE_TOAST: "REMOVE_TOAST",
} as const;

let count = 0;

function genId() {
  count = (count + 1) % Number.MAX_SAFE_INTEGER;
  return count.toString();
}

type ActionType = typeof actionTypes;

type Action =
  | {
      type: ActionType["ADD_TOAST"];
      toast: ToasterToast;
    }
  | {
      type: ActionType["UPDATE_TOAST"];
      toast: Partial<ToasterToast>;
    }
  | {
      type: ActionType["DISMISS_TOAST"];
      toastId?: ToasterToast["id"];
    }
  | {
      type: ActionType["REMOVE_TOAST"];
      toastId?: ToasterToast["id"];
    };

interface State {
  toasts: ToasterToast[];
}

const toastTimeouts = new Map<string, ReturnType<typeof setTimeout>>();

const addToRemoveQueue = (toastId: string) => {
  if (toastTimeouts.has(toastId)) {
    return;
  }

  const timeout = setTimeout(() => {
    toastTimeouts.delete(toastId);
    dispatch({
      type: "REMOVE_TOAST",
      toastId: toastId,
    });
  }, TOAST_REMOVE_DELAY);

  toastTimeouts.set(toastId, timeout);
};

export const reducer = (state: State, action: Action): State => {
  switch (action.type) {
    case "ADD_TOAST":
      return {
        ...state,
        toasts: [action.toast, ...state.toasts].slice(0, TOAST_LIMIT),
      };

    case "UPDATE_TOAST":
      return {
        ...state,
        toasts: state.toasts.map((t) => (t.id === action.toast.id ? { ...t, ...action.toast } : t)),
      };

    case "DISMISS_TOAST": {
      const { toastId } = action;

      // ! Side effects ! - This could be extracted into a dismissToast() action,
      // but I'll keep it here for simplicity
      if (toastId) {
        addToRemoveQueue(toastId);
      } else {
        state.toasts.forEach((toast) => {
          addToRemoveQueue(toast.id);
        });
      }

      return {
        ...state,
        toasts: state.toasts.map((t) =>
          t.id === toastId || toastId === undefined
            ? {
                ...t,
                open: false,
              }
            : t,
        ),
      };
    }
    case "REMOVE_TOAST":
      if (action.toastId === undefined) {
        return {
          ...state,
          toasts: [],
        };
      }
      return {
        ...state,
        toasts: state.toasts.filter((t) => t.id !== action.toastId),
      };
  }
};

const listeners: Array<(state: State) => void> = [];

let memoryState: State = { toasts: [] };

function dispatch(action: Action) {
  memoryState = reducer(memoryState, action);
  listeners.forEach((listener) => {
    listener(memoryState);
  });
}

type Toast = Omit<ToasterToast, "id">;

function toast({ ...props }: Toast) {
  const id = genId();

  const update = (props: ToasterToast) =>
    dispatch({
      type: "UPDATE_TOAST",
      toast: { ...props, id },
    });
  const dismiss = () => dispatch({ type: "DISMISS_TOAST", toastId: id });

  dispatch({
    type: "ADD_TOAST",
    toast: {
      ...props,
      id,
      open: true,
      onOpenChange: (open) => {
        if (!open) dismiss();
      },
    },
  });

  return {
    id: id,
    dismiss,
    update,
  };
}

function useToast() {
  const [state, setState] = React.useState<State>(memoryState);

  React.useEffect(() => {
    listeners.push(setState);
    return () => {
      const index = listeners.indexOf(setState);
      if (index > -1) {
        listeners.splice(index, 1);
      }
    };
  }, [state]);

  return {
    ...state,
    toast,
    dismiss: (toastId?: string) => dispatch({ type: "DISMISS_TOAST", toastId }),
  };
}

export { useToast, toast };
